const OrderTracker = ({ status, orderId }) => {
    const checkpoints = ["Order Placed", "Picked Up", "Dispatched", "Delivered"];
    const statusMap = {
        "placed": 0,
        "picked_up": 1,
        "dispatched": 2,
        "delivered": 3
    };
    const currentStatusIndex = statusMap[status] ?? -1;

    return (
        <div className="p-4 sm:p-6 bg-white rounded-xl shadow-lg">
            <h3 className="text-lg font-bold text-gray-800 mb-4">Order #{orderId.substring(0,8)}</h3>
            <div className="flex justify-between items-center">
                {checkpoints.map((point, index) => (
                    <React.Fragment key={point}>
                        <div className="flex flex-col items-center">
                            <div className={`w-8 h-8 sm:w-10 sm:h-10 rounded-full flex items-center justify-center text-white ${index <= currentStatusIndex ? 'bg-green-500' : 'bg-gray-300'}`}>
                                {index <= currentStatusIndex ? <CheckCheck size={20} /> : <Package size={20} />}
                            </div>
                            <p className={`mt-2 text-xs sm:text-sm text-center font-semibold ${index <= currentStatusIndex ? 'text-green-600' : 'text-gray-500'}`}>{point}</p>
                        </div>
                        {index < checkpoints.length - 1 && (
                            <div className={`flex-1 h-1 mx-2 ${index < currentStatusIndex ? 'bg-green-500' : 'bg-gray-300'}`}></div>
                        )}
                    </React.Fragment>
                ))}
            </div>
        </div>
    );
};


const OtpModal = ({ orderId, onVerify, onClose }) => {
    const [otp, setOtp] = useState('');
    const [error, setError] = useState('');

    const handleVerify = () => {
        if (otp.length === 6) {
            onVerify(orderId, otp);
            onClose();
        } else {
            setError('Please enter a 6-digit OTP.');
        }
    };
    
    return (
        <div className="fixed inset-0 bg-black bg-opacity-50 flex items-center justify-center z-50 p-4">
            <div className="bg-white rounded-xl p-8 shadow-2xl w-full max-w-sm">
                <h2 className="text-2xl font-bold text-gray-800 text-center mb-2">Complete Delivery</h2>
                <p className="text-center text-gray-500 mb-6">Enter the 6-digit OTP from the customer.</p>
                <input
                    type="text"
                    maxLength="6"
                    value={otp}
                    onChange={(e) => {
                        setOtp(e.target.value.replace(/[^0-9]/g, ''));
                        setError('');
                    }}
                    className="w-full p-4 text-center text-2xl tracking-widest font-mono border-2 border-gray-300 rounded-lg focus:ring-2 focus:ring-green-500 focus:border-green-500 transition"
                    placeholder="------"
                />
                {error && <p className="text-red-500 text-sm mt-2 text-center">{error}</p>}
                <div className="mt-6 flex flex-col sm:flex-row gap-3">
                    <button onClick={onClose} className="w-full bg-gray-200 text-gray-700 font-bold py-3 px-4 rounded-lg hover:bg-gray-300 transition-colors">Cancel</button>
                    <button onClick={handleVerify} className="w-full bg-green-500 text-white font-bold py-3 px-4 rounded-lg hover:bg-green-600 transition-colors">Verify OTP</button>
                </div>
            </div>
        </div>
    );
};

const AuthPage = ({ userType, onLogin, onRegister, onBack }) => {

  const [password, setPassword] = useState('');
  const [name, setName] = useState('');

    e.preventDefault();
    setError('');
    const result = isLogin ? onLogin(email, password) : onRegister(name, email, password);
    if (result && result.error) {
        setError(result.error);
    }
  };

  return (
    <div className="min-h-screen bg-gray-50 flex flex-col justify-center items-center p-4">

                    {isLogin ? 'Sign in' : 'Create an account'} as a {userType}

            </div>
            <form className="mt-8 space-y-6 bg-white p-8 rounded-xl shadow-lg" onSubmit={handleSubmit}>
                {!isLogin && (

// --- CUSTOMER INTERFACE ---
const CustomerInterface = ({ user, onLogout, allProducts, orders, setOrders }) => {
    const [activeTab, setActiveTab] = useState('store');
    const [cart, setCart] = useState([]);
    const [currentUser, setCurrentUser] = useState(user);

    const handleAddToCart = (product) => {
        setCart(prevCart => {
            const existingItem = prevCart.find(item => item.id === product.id);
            if (existingItem) {
                return prevCart.map(item => item.id === product.id ? { ...item, quantity: item.quantity + 1 } : item);
            }
            return [...prevCart, { ...product, quantity: 1 }];
        });
    };
    
    const updateCartQuantity = (productId, newQuantity) => {
        if (newQuantity <= 0) {
            setCart(cart.filter(item => item.id !== productId));
        } else {
            setCart(cart.map(item => item.id === productId ? { ...item, quantity: newQuantity } : item));
        }
    };

    const placeOrder = (finalAmount, loyaltyPointsEarned) => {
        if (cart.length === 0) return;
        const newOrder = {
            id: `order_${Date.now()}`,
            customerId: currentUser.id,
            customerName: currentUser.name,
            items: cart,
            totalAmount: finalAmount,
            address: currentUser.addresses[0].text,
            status: 'placed',
            otp: Math.floor(100000 + Math.random() * 900000).toString(),
            createdAt: new Date().toISOString()
        };
        setOrders(prevOrders => [newOrder, ...prevOrders]);
        
        // Update user's loyalty points
        setCurrentUser(prevUser => ({
            ...prevUser,
            loyaltyPoints: prevUser.loyaltyPoints + loyaltyPointsEarned
        }));
        
        setCart([]);
        setActiveTab('status');
    };

    const NavButton = ({ tabName, icon, label }) => (
        <button
            onClick={() => setActiveTab(tabName)}
            className={`flex-1 sm:flex-none flex flex-col sm:flex-row items-center justify-center gap-2 px-4 py-2 rounded-lg transition-colors duration-200 ${activeTab === tabName ? 'bg-green-500 text-white shadow-md' : 'text-gray-600 hover:bg-green-100'}`}
        >
            {icon}
            <span className="text-sm font-semibold">{label}</span>
        </button>
    );

    const CustomerHeader = () => (
        <header className="bg-white shadow-md sticky top-0 z-10 p-4">
            <div className="container mx-auto flex justify-between items-center">
                <div className="flex items-center gap-2">
                    <UtensilsCrossed className="text-green-500" size={32} />
                    <h1 className="text-2xl font-bold text-gray-800">FresHunt</h1>
                </div>
                 <div className="flex items-center gap-4">
                    <div className="flex items-center gap-2 text-yellow-500">
                        <Star size={20} />
                        <span className="font-bold">{currentUser.loyaltyPoints}</span>
                    </div>
                    <div className="text-right hidden sm:block">
                        <p className="font-semibold">{currentUser.name}</p>
                        <p className="text-xs text-gray-500">{currentUser.email}</p>
                    </div>
                    <CircleUserRound size={32} className="text-gray-600" />
                    <button onClick={onLogout} className="bg-red-500 text-white px-3 py-2 rounded-lg font-semibold hover:bg-red-600 transition-colors">Logout</button>
                </div>
            </div>
            <nav className="container mx-auto mt-4 flex justify-center sm:justify-start gap-2 sm:gap-4 bg-gray-100 p-2 rounded-lg">
                <NavButton tabName="store" icon={<ChefHat size={20}/>} label="Store" />
                <NavButton tabName="cart" icon={<ShoppingCart size={20}/>} label={`My Cart (${cart.reduce((sum, item) => sum + item.quantity, 0)})`} />
                <NavButton tabName="status" icon={<Package size={20}/>} label="Order Status" />
            </nav>
        </header>
    );

    const StoreView = () => (
        <div className="p-4 sm:p-6">
            {Object.entries(allProducts).map(([category, products]) => (
                <div key={category} className="mb-10">
                    <h2 className="text-3xl font-bold text-gray-800 capitalize mb-6 border-l-4 border-green-500 pl-4">{category}</h2>
                    <div className="grid grid-cols-1 sm:grid-cols-2 md:grid-cols-3 lg:grid-cols-4 xl:grid-cols-5 gap-6">
                        {products.map(p => <ProductCard key={p.id} product={p} onAddToCart={handleAddToCart} />)}
                    </div>
                </div>
            ))}
        </div>
    );
    
    const CartView = () => {
        const [pointsToUse, setPointsToUse] = useState(0);
        const subtotal = cart.reduce((sum, item) => sum + item.price * item.quantity, 0);
        const discount = Math.min(pointsToUse, subtotal);
        const finalAmount = subtotal - discount;
        const loyaltyPointsEarned = Math.floor(finalAmount / 100) * 5;

        useEffect(() => {
            setPointsToUse(0);
        }, [cart]);

        const handlePlaceOrder = () => {
             // Deduct used points
            const newLoyaltyPoints = currentUser.loyaltyPoints - Math.floor(discount);
            setCurrentUser(prev => ({...prev, loyaltyPoints: newLoyaltyPoints }));
            placeOrder(finalAmount, loyaltyPointsEarned);
        }

        return (
             <div className="p-4 sm:p-6">
                <h2 className="text-3xl font-bold text-gray-800 mb-6">My Cart</h2>
                {cart.length === 0 ? (
                    <div className="text-center py-20 bg-white rounded-lg shadow">
                        <ShoppingCart size={60} className="mx-auto text-gray-300" />
                        <p className="mt-4 text-gray-500 text-lg">Your cart is empty.</p>
                        <button onClick={() => setActiveTab('store')} className="mt-6 bg-green-500 text-white font-bold py-3 px-6 rounded-lg hover:bg-green-600 transition-colors">
                            Start Shopping
                        </button>
                    </div>
                ) : (
                    <div className="grid grid-cols-1 lg:grid-cols-3 gap-8">
                        <div className="lg:col-span-2 bg-white p-6 rounded-xl shadow-lg">
                           <h3 className="text-xl font-semibold mb-4">Items</h3>
                            {cart.map(item => (
                                <div key={item.id} className="flex items-center justify-between py-4 border-b">
                                    <div className="flex items-center gap-4">
                                        <div className="text-4xl">{item.image}</div>
                                        <div>
                                            <p className="font-semibold text-lg">{item.name}</p>
                                            <p className="text-gray-500">₹{item.price.toFixed(2)}</p>
                                        </div>
                                    </div>
                                    <div className="flex items-center gap-3">
                                        <button onClick={() => updateCartQuantity(item.id, item.quantity - 1)} className="w-8 h-8 rounded-full bg-gray-200 hover:bg-gray-300">-</button>
                                        <span className="font-bold text-lg w-8 text-center">{item.quantity}</span>
                                        <button onClick={() => updateCartQuantity(item.id, item.quantity + 1)} className="w-8 h-8 rounded-full bg-gray-200 hover:bg-gray-300">+</button>
                                    </div>
                                    <p className="font-bold text-lg w-24 text-right">₹{(item.price * item.quantity).toFixed(2)}</p>
                                </div>
                            ))}
                        </div>
                        <div className="bg-white p-6 rounded-xl shadow-lg h-fit">
                            <h3 className="text-xl font-semibold mb-4 border-b pb-3">Order Summary</h3>
                            <div className="space-y-3">
                                <div className="flex justify-between"><span>Subtotal</span> <span className="font-semibold">₹{subtotal.toFixed(2)}</span></div>
                                <div className="flex justify-between"><span>Discount</span> <span className="font-semibold text-green-600">- ₹{discount.toFixed(2)}</span></div>
                                <div className="flex justify-between text-xl font-bold border-t pt-3"><span>Total</span> <span>₹{finalAmount.toFixed(2)}</span></div>
                            </div>

                            <div className="mt-6 bg-yellow-50 p-4 rounded-lg">
                                <h4 className="font-semibold text-yellow-800">Use Loyalty Points</h4>
                                <p className="text-sm text-yellow-700">Available: {currentUser.loyaltyPoints} points</p>
                                <div className="mt-2">
                                    <input 
                                        type="range"
                                        min="0"
                                        max={Math.min(currentUser.loyaltyPoints, subtotal)}
                                        value={pointsToUse}
                                        onChange={(e) => setPointsToUse(Number(e.target.value))}
                                        className="w-full h-2 bg-yellow-200 rounded-lg appearance-none cursor-pointer"
                                    />
                                    <div className="flex justify-between text-xs mt-1">
                                        <span>0</span>
                                        <span>{Math.min(currentUser.loyaltyPoints, subtotal)}</span>
                                    </div>
                                </div>
                                <p className="text-center font-bold mt-2 text-yellow-800">{pointsToUse} points used</p>
                            </div>
                            
                            <p className="text-center text-sm text-green-600 mt-4 font-semibold">You will earn {loyaltyPointsEarned} loyalty points for this order.</p>

                            <button onClick={handlePlaceOrder} className="mt-6 w-full bg-green-500 text-white font-bold py-3 rounded-lg text-lg hover:bg-green-600 transition-colors">
                                Place Order
                            </button>
                        </div>
                    </div>
                )}
            </div>
        );
    };

    const StatusView = () => {
        const myOrders = orders.filter(o => o.customerId === currentUser.id);
        const activeOrder = myOrders.find(o => o.status !== 'delivered');
        const orderHistory = myOrders.filter(o => o.status === 'delivered');

        return (
            <div className="p-4 sm:p-6">
                <h2 className="text-3xl font-bold text-gray-800 mb-6">Order Status</h2>
                 {activeOrder ? (
                    <div className="bg-white p-6 rounded-xl shadow-lg mb-8">
                        <h3 className="text-xl font-bold text-gray-800 mb-4">Current Order Tracking</h3>
                        <OrderTracker status={activeOrder.status} orderId={activeOrder.id} />
                        <div className="grid md:grid-cols-3 gap-6 mt-6 pt-6 border-t">
                             <div>
                                <h4 className="font-bold text-gray-600">Items</h4>
                                <ul className="list-disc list-inside mt-2 text-gray-700">
                                   {activeOrder.items.map(item => (
                                        <li key={item.id}>{item.name} x {item.quantity}</li>
                                    ))}
                                </ul>
                             </div>
                              <div>
                                <h4 className="font-bold text-gray-600">Delivery Address</h4>
                                <p className="mt-2 text-gray-700">{activeOrder.address}</p>
                             </div>
                             <div className="bg-green-100 border-2 border-dashed border-green-400 p-4 rounded-lg text-center">
                                <h4 className="font-bold text-green-800">Delivery OTP</h4>
                                <p className="text-3xl font-extrabold tracking-widest text-green-700 mt-2">{activeOrder.otp}</p>
                                <p className="text-xs text-green-600 mt-1">Share this with the delivery person</p>
                             </div>
                        </div>
                    </div>
                ) : (
                    <div className="text-center py-10 bg-white rounded-lg shadow mb-8">
                        <Package size={48} className="mx-auto text-gray-300" />
                        <p className="mt-4 text-gray-500">You have no active orders.</p>
                    </div>
                )}
                
                <div>
                    <h3 className="text-2xl font-bold text-gray-800 mb-4 flex items-center gap-2"><History/> Order History</h3>
                    <div className="space-y-4">
                        {orderHistory.length > 0 ? orderHistory.map(order => (
                             <div key={order.id} className="bg-white p-4 rounded-lg shadow-sm flex justify-between items-center">
                                <div>
                                    <p className="font-bold">Order #{order.id.substring(0,8)}</p>
                                    <p className="text-sm text-gray-500">{new Date(order.createdAt).toLocaleString()}</p>
                                </div>
                                <p className="font-bold text-lg">₹{order.totalAmount.toFixed(2)}</p>
                                <span className="px-3 py-1 bg-green-200 text-green-800 text-sm font-semibold rounded-full">Delivered</span>
                             </div>
                        )) : <p className="text-gray-500">No past orders found.</p>}
                    </div>
                </div>
            </div>
        );
    };

    return (
        <div className="bg-gray-50 min-h-screen">
            <CustomerHeader />
            <main className="container mx-auto">
                {activeTab === 'store' && <StoreView />}
                {activeTab === 'cart' && <CartView />}
                {activeTab === 'status' && <StatusView />}
            </main>
        </div>
    );
};

// --- DELIVERY INTERFACE ---
const DeliveryInterface = ({ user, onLogout, orders, setOrders }) => {
    const [activeTab, setActiveTab] = useState('new');
    const [showOtpModal, setShowOtpModal] = useState(false);
    const [selectedOrder, setSelectedOrder] = useState(null);

    const handleAcceptOrder = (orderId) => {
        setOrders(orders.map(o => o.id === orderId ? { ...o, status: 'picked_up', deliveryPersonId: user.id } : o));
    };

    const handleUpdateStatus = (orderId, newStatus) => {
        setOrders(orders.map(o => o.id === orderId ? { ...o, status: newStatus } : o));
    };
    
    const handleDeliverClick = (order) => {
        setSelectedOrder(order);
        setShowOtpModal(true);
    };

    const handleVerifyOtp = (orderId, otp) => {
        const order = orders.find(o => o.id === orderId);
        if (order && order.otp === otp) {
            handleUpdateStatus(orderId, 'delivered');
            alert('Order delivered successfully!');
        } else {
            alert('Invalid OTP. Please try again.');
        }
    };
    
    const OrderCard = ({ order, children }) => (
        <div className="bg-white p-5 rounded-xl shadow-lg">
            <div className="flex justify-between items-start">
                <div>
                    <p className="font-bold text-lg text-gray-800">Order #{order.id.substring(0,8)}</p>
                    <p className="text-sm text-gray-500">For: {order.customerName}</p>
                </div>
                <p className="font-bold text-xl text-green-600">₹{order.totalAmount.toFixed(2)}</p>
            </div>
            <div className="mt-4 pt-4 border-t">
                <div className="flex items-start gap-3">
                    <MapPin className="text-gray-400 mt-1" size={20}/>
                    <div>
                        <p className="font-semibold">Delivery Address</p>
                        <p className="text-gray-600">{order.address}</p>
                    </div>
                </div>
                 <div className="flex items-start gap-3 mt-3">
                    <ShoppingCart className="text-gray-400 mt-1" size={20}/>
                    <div>
                        <p className="font-semibold">Items</p>
                        <p className="text-gray-600 text-sm">
                            {order.items.map(i => `${i.name} (x${i.quantity})`).join(', ')}
                        </p>
                    </div>
                </div>
            </div>
            <div className="mt-5">{children}</div>
        </div>
    );
    
    const NavButton = ({ tabName, icon, label, count }) => (
        <button
            onClick={() => setActiveTab(tabName)}
            className={`flex-1 sm:flex-none flex items-center justify-center gap-2 px-4 py-3 rounded-lg transition-colors duration-200 relative ${activeTab === tabName ? 'bg-green-500 text-white shadow-md' : 'text-gray-600 hover:bg-green-100'}`}
        >
            {icon}
            <span className="font-semibold">{label}</span>
            {count > 0 && <span className="absolute -top-2 -right-2 bg-red-500 text-white text-xs w-6 h-6 rounded-full flex items-center justify-center">{count}</span>}
        </button>
    );

    const newOrders = orders.filter(o => o.status === 'placed');
    const acceptedOrders = orders.filter(o => o.deliveryPersonId === user.id && (o.status === 'picked_up' || o.status === 'dispatched'));
    const deliveredOrders = orders.filter(o => o.deliveryPersonId === user.id && o.status === 'delivered');

    return (
        <div className="bg-gray-100 min-h-screen">
            <header className="bg-white shadow-md sticky top-0 z-10 p-4">
                <div className="container mx-auto flex justify-between items-center">
                    <div className="flex items-center gap-2">
                        <Truck className="text-green-500" size={32} />
                        <h1 className="text-2xl font-bold text-gray-800">Delivery Panel</h1>
                    </div>
                    <div className="flex items-center gap-4">
                        <div>
                            <p className="font-semibold">{user.name}</p>
                        </div>
                        <button onClick={onLogout} className="bg-red-500 text-white px-3 py-2 rounded-lg font-semibold hover:bg-red-600 transition-colors">Logout</button>
                    </div>
                </div>
            </header>
            <nav className="container mx-auto mt-4 flex justify-center sm:justify-start gap-2 sm:gap-4 bg-white p-2 rounded-lg shadow-sm">
                <NavButton tabName="new" icon={<PackageOpen size={20}/>} label="New Orders" count={newOrders.length}/>
                <NavButton tabName="accepted" icon={<CheckCheck size={20}/>} label="Accepted" count={acceptedOrders.length}/>
                <NavButton tabName="delivered" icon={<History size={20}/>} label="Delivered" count={deliveredOrders.length}/>
            </nav>
            <main className="container mx-auto p-4 sm:p-6">
                 <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">
                    {activeTab === 'new' && (newOrders.length > 0 ? newOrders.map(order => (
                        <OrderCard key={order.id} order={order}>
                            <button onClick={() => handleAcceptOrder(order.id)} className="w-full bg-green-500 text-white font-bold py-3 rounded-lg hover:bg-green-600 transition-colors">Accept Order</button>
                        </OrderCard>
                    )) : <p className="text-gray-500 md:col-span-2 lg:col-span-3 text-center">No new orders available.</p>)}

                    {activeTab === 'accepted' && (acceptedOrders.length > 0 ? acceptedOrders.map(order => (
                        <OrderCard key={order.id} order={order}>
                            <div className="flex flex-col gap-3">
                                <button
                                    disabled={order.status !== 'picked_up'}
                                    onClick={() => handleUpdateStatus(order.id, 'dispatched')}
                                    className="w-full py-3 rounded-lg font-bold transition-colors disabled:bg-gray-300 disabled:cursor-not-allowed bg-blue-500 text-white hover:bg-blue-600"
                                >
                                    Mark as Dispatched
                                </button>
                                <button
                                    disabled={order.status !== 'dispatched'}
                                    onClick={() => handleDeliverClick(order)}
                                    className="w-full py-3 rounded-lg font-bold transition-colors disabled:bg-gray-300 disabled:cursor-not-allowed bg-green-500 text-white hover:bg-green-600"
                                >
                                    Mark as Delivered
                                </button>
                            </div>
                        </OrderCard>
                    )) : <p className="text-gray-500 md:col-span-2 lg:col-span-3 text-center">No active orders.</p>)}

                    {activeTab === 'delivered' && (deliveredOrders.length > 0 ? deliveredOrders.map(order => (
                        <OrderCard key={order.id} order={order}>
                             <div className="text-center p-3 bg-green-100 text-green-800 font-bold rounded-lg">
                                Delivered on {new Date(order.createdAt).toLocaleDateString()}
                            </div>
                        </OrderCard>
                    )) : <p className="text-gray-500 md:col-span-2 lg:col-span-3 text-center">No delivered orders yet.</p>)}
                </div>
            </main>
            {showOtpModal && selectedOrder && (
                <OtpModal 
                    orderId={selectedOrder.id} 
                    onVerify={handleVerifyOtp} 
                    onClose={() => setShowOtpModal(false)} 
                />
            )}
        </div>
    );
};

// --- MAIN APP COMPONENT ---
export default function App() {
    const [userType, setUserType] = useState(null); // 'customer', 'delivery', or null
    const [currentUser, setCurrentUser] = useState(null);
    const [users, setUsers] = useState(initialUsers);
    
    // This state is the single source of truth for all orders, connecting both interfaces.
    const [orders, setOrders] = useState([]);
    
    // Simulate fetching initial orders on load
    useEffect(() => {
        // In a real app, you would fetch this from your database.
        // Adding a sample order for demonstration.
        if (orders.length === 0) {
             setOrders([
                {
                    id: `order_init`,
                    customerId: 'cust1',
                    customerName: 'Suresh Kumar',
                    items: [{ id: 1, name: 'Tomato', price: 40, image: '🍅', quantity: 2}, {id: 11, name: 'Apple', price: 120, image: '🍎', quantity: 1}],
                    totalAmount: 200,
                    address: '123, MG Road, Bangalore',
                    status: 'placed',
                    otp: '123456',
                    createdAt: new Date().toISOString()
                }
             ]);
        }
    }, []);

    const handleLogin = (email, password) => {
        const userPool = userType === 'customer' ? users.customers : users.delivery;
        const foundUser = userPool.find(u => u.email === email && u.password === password);
        if (foundUser) {
            setCurrentUser(foundUser);
            return { success: true };
        }
        return { error: 'Invalid email or password.' };
    };

    const handleRegister = (name, email, password) => {
        if (userType === 'customer') {
            if (users.customers.find(u => u.email === email)) {
                return { error: 'User with this email already exists.' };
            }
            const newUser = { id: `cust_${Date.now()}`, name, email, password, loyaltyPoints: 0, addresses: [{id: 1, text: 'Please add an address'}] };
            setUsers(prev => ({...prev, customers: [...prev.customers, newUser]}));
            setCurrentUser(newUser);
        } else {
             if (users.delivery.find(u => u.email === email)) {
                return { error: 'User with this email already exists.' };
            }
            const newUser = { id: `del_${Date.now()}`, name, email, password };
            setUsers(prev => ({...prev, delivery: [...prev.delivery, newUser]}));
            setCurrentUser(newUser);
        }
        return { success: true };
    };

    const handleLogout = () => {
        setCurrentUser(null);
    };

    const handleBackToLanding = () => {
        setUserType(null);
        setCurrentUser(null);
    };

    if (!userType) {
        return <LandingPage onSelectUserType={setUserType} />;
    }

    if (!currentUser) {
        return <AuthPage 
            userType={userType} 
            onLogin={handleLogin}
            onRegister={handleRegister}
            onBack={handleBackToLanding}
        />;
    }
    
    if (userType === 'customer') {
        return <CustomerInterface user={currentUser} onLogout={handleLogout} allProducts={initialProducts} orders={orders} setOrders={setOrders} />;
    }
    
    if (userType === 'delivery') {
        return <DeliveryInterface user={currentUser} onLogout={handleLogout} orders={orders} setOrders={setOrders} />;
    }

    return <div>Error: Invalid state.</div>;
}
